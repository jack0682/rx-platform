cmake_minimum_required(VERSION 3.22)
project(rxclcpp VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/rxclcpp/build_info.hpp")
  message(FATAL_ERROR "Use tools/prepare_clients.py to prepare this package from a verified bundle first")
endif()
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
set(RX_PROTOCOL_BUNDLE "${CMAKE_CURRENT_SOURCE_DIR}/../protocol" CACHE PATH "Prepared content-addressed RX source bundle")
set(RX_PROTO_ROOT "${RX_PROTOCOL_BUNDLE}")
include(cmake/VerifyBundle.cmake)
set(RX_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${RX_GENERATED}")
file(GLOB_RECURSE RX_RPC_PROTO RELATIVE "${RX_PROTO_ROOT}" "${RX_PROTO_ROOT}/rx/*.proto")
set(RX_MESSAGE_SOURCES)
foreach(PROTO IN LISTS RX_RPC_PROTO)
  string(REPLACE ".proto" ".pb.cc" CPP "${PROTO}")
  string(REPLACE ".proto" ".pb.h" HEADER "${PROTO}")
  add_custom_command(OUTPUT "${RX_GENERATED}/${CPP}" "${RX_GENERATED}/${HEADER}"
    COMMAND protobuf::protoc --cpp_out=${RX_GENERATED} --proto_path=${RX_PROTO_ROOT} "${RX_PROTO_ROOT}/${PROTO}"
    DEPENDS "${RX_PROTO_ROOT}/${PROTO}" "${RX_PROTO_ROOT}/rx/contract/v1/contract.proto" VERBATIM)
  list(APPEND RX_MESSAGE_SOURCES "${RX_GENERATED}/${CPP}")
endforeach()
add_library(rxclcpp SHARED src/wire.cpp src/client.cpp ${RX_MESSAGE_SOURCES})
add_library(rxclcpp::rxclcpp ALIAS rxclcpp)
set_target_properties(rxclcpp PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
target_compile_features(rxclcpp PUBLIC cxx_std_20)
target_compile_options(rxclcpp PRIVATE -Wall -Wextra -Werror)
target_include_directories(rxclcpp PUBLIC $<BUILD_INTERFACE:${RX_GENERATED}> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(rxclcpp PUBLIC protobuf::libprotobuf gRPC::grpc++)
install(TARGETS rxclcpp EXPORT rxclcppTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY "${RX_GENERATED}/rx" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.pb.h")
install(DIRECTORY "${RX_PROTO_ROOT}/" DESTINATION ${CMAKE_INSTALL_DATADIR}/rxclcpp/protocol)
install(EXPORT rxclcppTargets NAMESPACE rxclcpp:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rxclcpp)
configure_package_config_file(cmake/rxclcppConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/rxclcppConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rxclcpp)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/rxclcppConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/rxclcppConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/rxclcppConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rxclcpp)
include(CTest)
if(BUILD_TESTING)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(OpenSSL REQUIRED)
  add_custom_command(OUTPUT "${RX_GENERATED}/strict-wire-v1/probe.pb.cc" "${RX_GENERATED}/strict-wire-v1/probe.pb.h"
    COMMAND protobuf::protoc --cpp_out=${RX_GENERATED} --proto_path=${RX_PROTO_ROOT} "${RX_PROTO_ROOT}/strict-wire-v1/probe.proto"
    DEPENDS "${RX_PROTO_ROOT}/strict-wire-v1/probe.proto" VERBATIM)
  add_executable(rxclcpp-conformance tests/conformance.cpp "${RX_GENERATED}/strict-wire-v1/probe.pb.cc")
  target_link_libraries(rxclcpp-conformance PRIVATE rxclcpp nlohmann_json::nlohmann_json OpenSSL::Crypto)
  target_compile_options(rxclcpp-conformance PRIVATE -Wall -Wextra -Werror)
  add_test(NAME strict-wire-v1 COMMAND rxclcpp-conformance "${RX_PROTO_ROOT}/strict-wire-v1/vectors.json")
endif()
